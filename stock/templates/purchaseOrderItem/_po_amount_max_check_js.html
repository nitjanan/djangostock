//ปุ่มบันทึกและส่งอนุมัติ: จำนวนเงินรวมทั้งสิ้นตั้งแต่วงเงินของสิทธิอนุมัติใบสั่งซื้อ (CAAPO ap_amount_max) ขึ้นไป //แจ้งเตือนและไม่ให้บันทึก (ต้องออกใบเปรียบเทียบราคาก่อน) server เช็คซ้ำใน check_po_amount_over_max //ปุ่มไม่มี data-amount-max (ใบที่ออกจากใบเปรียบเทียบ/ไม่ได้ตั้งวงเงิน) จะไม่เช็ค คืน true ถ้าส่งต่อได้ function checkPOAmountOverMax(button) { var max = parseFloat($(button).attr('data-amount-max')); if (isNaN(max)) return true; var amount = parseFloat(String($('#id_amount').val() || '').replace(/,/g, '')) || 0; if (amount < max) return true; //วงเงินเต็มจำนวนแสดงไม่มีทศนิยม (5,000) ถ้ามีเศษสตางค์แสดง 2 ตำแหน่ง var digits = Number.isInteger(max) ? 0 : 2; var maxText = max.toLocaleString('th-TH', { minimumFractionDigits: digits, maximumFractionDigits: digits }); alert('ไม่สามารถบันทึกและส่งอนุมัติได้\nเนื่องจากยอดเงินตั้งแต่ ' + maxText + ' บาทขึ้นไปต้องมีใบเปรียบเทียบราคา กรุณาออกใบเปรียบเทียบราคาก่อนออกใบสั่งซื้อ'); return false; }
